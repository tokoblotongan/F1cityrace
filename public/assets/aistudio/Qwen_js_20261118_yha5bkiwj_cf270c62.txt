// Scene, Mediterranean lighting + sky/env bake, bloom pipeline,
// Chase / T-Cam / Cockpit camera switcher ('C'), keyboard input
// (WASD/Arrows, Space/Shift = DRS, 'R' = reset), fixed-step 60 FPS loop.

// Expected importmap in index.html (Step: UI shell):
// <script type="importmap">
// { "imports": {

import * as THREE from 'three';
import { Sky } from 'three/addons/objects/Sky.js';
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';
import { OutputPass } from 'three/addons/postprocessing/OutputPass.js';

import { buildMonacoTrack } from './track.js';
import { createF1Car, updateCarPhysics } from './carPhysics.js';

// === INTEGRATION HOOK: aiRacer.js (Step 2) ===================================
// import { initAIRacers, updateAIRacers } from './aiRacer.js';
// After buildMonacoTrack:   const aiList = initAIRacers(scene, trackData);
// Inside physics substep:   const raceState = updateAIRacers(aiList, trackData, STEP);
// raceState feeds updateHUD(raceState) below and drives
// trackData.startFinishLine.lights[] during the start-light sequence.
// =============================================================================

// === INTEGRATION HOOK: audioHud.js (Step 3) ==================================
// import { initAudio, updateAudio, updateHUD } from './audioHud.js';
// On first user gesture:    initAudio();               (autoplay policy)
// Each render frame:        updateAudio(carState);     (V6 engine synth)
//                           updateHUD(raceState);      (broadcast DOM HUD)
// =============================================================================

const STEP = 1 / 120;               // fixed physics substep (s)
const MAX_SUBSTEPS = 6;
const CAMERA_MODES = ['CHASE', 'T-CAM', 'COCKPIT'];

// --- renderer / canvas ---------------------------------------------------------
let canvas = document.getElementById('game-canvas');
if (!canvas) {
  canvas = document.createElement('canvas');
  canvas.id = 'game-canvas';
  canvas.style.cssText = 'position:fixed;inset:0;width:100%;height:100%;display:block;';
  document.body.appendChild(canvas);
}

let renderer;
try {
  renderer = new THREE.WebGLRenderer({ canvas, antialias: true, powerPreference: 'high-performance' });
} catch (e) {
  const msg = document.createElement('div');
  msg.style.cssText = 'position:fixed;inset:0;display:grid;place-items:center;background:#0d1117;color:#f2efe6;font:600 18px system-ui;text-align:center;padding:24px;';
  msg.textContent = 'WebGL is unavailable on this device — Monaco GP cannot start.';
  document.body.appendChild(msg);
  throw e;
}
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
renderer.setSize(window.innerWidth, window.innerHeight);
renderer.outputColorSpace = THREE.SRGBColorSpace;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.05;
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;

// --- scene / atmosphere ----------------------------------------------------------
const scene = new THREE.Scene();
const DAY_FOG = new THREE.Color(0xcfe3ef);
const TUNNEL_FOG = new THREE.Color(0x0d0f12);
scene.fog = new THREE.FogExp2(DAY_FOG.clone(), 0.00135);

const camera = new THREE.PerspectiveCamera(72, window.innerWidth / window.innerHeight, 0.1, 3200);
camera.position.set(0, 6, 14);

// Sky + baked environment (PMREM once — never recomputed per frame)
const sunDir = new THREE.Vector3();
{
  const elev = THREE.MathUtils.degToRad(38), azim = THREE.MathUtils.degToRad(150);
  sunDir.setFromSphericalCoords(1, Math.PI / 2 - elev, azim);
}
const sky = new Sky();
sky.scale.setScalar(4500);
sky.material.uniforms.turbidity.value = 5;
sky.material.uniforms.rayleigh.value = 1.4;
sky.material.uniforms.mieCoefficient.value = 0.004;
sky.material.uniforms.mieDirectionalG.value = 0.85;
sky.material.uniforms.sunPosition.value.copy(sunDir);
scene.add(sky);
{
  const envScene = new THREE.Scene();
  const envSky = new Sky();
  envSky.scale.setScalar(4500);
  envSky.material.uniforms.turbidity.value = 5;
  envSky.material.uniforms.rayleigh.value = 1.4;
  envSky.material.uniforms.mieCoefficient.value = 0.004;
  envSky.material.uniforms.mieDirectionalG.value = 0.85;
  envSky.material.uniforms.sunPosition.value.copy(sunDir);
  envScene.add(envSky);
  const pmrem = new THREE.PMREMGenerator(renderer);
  scene.environment = pmrem.fromScene(envScene, 0.04).texture;
  pmrem.dispose();
}

// Lights: warm key sun (shadow frustum follows the car) + cool hemi fill
const hemi = new THREE.HemisphereLight(0xbdd7ff, 0x6f6a5e, 0.5);
scene.add(hemi);
const sun = new THREE.DirectionalLight(0xfff1d6, 3.2);
sun.castShadow = true;
sun.shadow.mapSize.set(2048, 2048);
sun.shadow.camera.left = -140; sun.shadow.camera.right = 140;
sun.shadow.camera.top = 140; sun.shadow.camera.bottom = -140;
sun.shadow.camera.near = 50; sun.shadow.camera.far = 900;
sun.shadow.bias = -0.0004;
sun.shadow.normalBias = 0.5;
scene.add(sun, sun.target);

// --- world + player ------------------------------------------------------------------
const trackData = buildMonacoTrack(scene);
const playerCar = createF1Car(0xe10600, true);   // Rosso Monaco
scene.add(playerCar);
updateCarPhysics(playerCar, { reset: true }, trackData, 0);

// aiRacer hook point: const aiList = initAIRacers(scene, trackData);

const waterMat = trackData.trackCurve.userData.waterMaterial;
waterMat.uniforms.uSunDir.value.copy(sunDir);
const tunnel = trackData.trackCurve.userData.tunnel;

// --- post pipeline: subtle bloom for emissives (tunnel, ERS, DRS strips) ---------------
const composer = new EffectComposer(renderer);
composer.addPass(new RenderPass(scene, camera));
const bloom = new UnrealBloomPass(
  new THREE.Vector2(window.innerWidth, window.innerHeight), 0.32, 0.55, 0.82);
composer.addPass(bloom);
composer.addPass(new OutputPass());

// --- input ------------------------------------------------------------------------------
const keys = Object.create(null);
let resetQueued = false;
let cameraMode = 0;
let audioStarted = false;

window.addEventListener('keydown', (e) => {
  if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space'].includes(e.code)) e.preventDefault();
  if (e.repeat) return;
  keys[e.code] = true;
  if (e.code === 'KeyC') setCameraMode((cameraMode + 1) % CAMERA_MODES.length);
  if (e.code === 'KeyR') resetQueued = true;
  if (!audioStarted) {
    audioStarted = true;
    // audioHud hook: initAudio();  ← first-gesture unlock (Web Audio autoplay policy)
  }
});
window.addEventListener('keyup', (e) => { keys[e.code] = false; });
window.addEventListener('blur', () => { for (const k in keys) keys[k] = false; });

// smoothed analog controls from digital keys
const controls = { throttle: 0, brake: 0, steer: 0, drs: false, reset: false };
function buildControls(dt) {
  const tT = (keys.KeyW || keys.ArrowUp) ? 1 : 0;
  const bT = (keys.KeyS || keys.ArrowDown) ? 1 : 0;
  const sT = (keys.KeyA || keys.ArrowLeft) ? 1 : (keys.KeyD || keys.ArrowRight) ? -1 : 0;
  controls.throttle += (tT - controls.throttle) * Math.min(1, (tT > controls.throttle ? 5 : 9) * dt);
  controls.brake += (bT - controls.brake) * Math.min(1, 9 * dt);
  controls.steer += (sT - controls.steer) * Math.min(1, (sT !== 0 ? 3.4 : 6) * dt);
  controls.drs = !!(keys.Space || keys.ShiftLeft || keys.ShiftRight);
  controls.reset = resetQueued;
  resetQueued = false;
}

// --- camera controller ---------------------------------------------------------------------
const chasePos = new THREE.Vector3().copy(playerCar.position).add(new THREE.Vector3(0, 4, 10));
const _fwd = new THREE.Vector3();
const _look = new THREE.Vector3();
const _off = new THREE.Vector3();
const TCAM_OFFSET = new THREE.Vector3(0, 1.3, 0.18);
const COCKPIT_OFFSET = new THREE.Vector3(0, 0.92, -0.14);
let baseFov = 72;

function setCameraMode(m) {
  cameraMode = m;
  const cockpit = CAMERA_MODES[m] === 'COCKPIT';
  playerCar.userData.parts.helmet.visible = !cockpit;
  camera.near = cockpit ? 0.06 : 0.1;
  camera.updateProjectionMatrix();
}

function updateCamera(dt, carState, elapsed) {
  const P = playerCar.userData.physics;
  _fwd.set(-Math.sin(P.yaw), 0, -Math.cos(P.yaw));
  const speedFrac = Math.min(1, carState.speedKmh / 330);
  const shake = carState.hitIntensity * 0.35 + speedFrac * 0.02 + (carState.onCurb ? 0.035 : 0);
  const sx = Math.sin(elapsed * 47.3) * shake, sy = Math.cos(elapsed * 39.1) * shake * 0.7;

  const mode = CAMERA_MODES[cameraMode];
  if (mode === 'CHASE') {
    _off.copy(playerCar.position)
      .addScaledVector(_fwd, -(9.5 + speedFrac * 1.8))
      .add(_look.set(0, 3.4 + speedFrac * 0.5, 0));
    chasePos.lerp(_off, 1 - Math.exp(-6 * dt));
    camera.position.copy(chasePos).add(_look.set(sx, sy, 0));
    _look.copy(playerCar.position).addScaledVector(_fwd, 6).add(new THREE.Vector3(0, 1.1, 0));
    camera.lookAt(_look);
    baseFov = 70 + speedFrac * 10 + (carState.drsActive ? 3 : 0);
  } else {
    const off = mode === 'T-CAM' ? TCAM_OFFSET : COCKPIT_OFFSET;
    camera.position.copy(playerCar.localToWorld(_off.copy(off)));
    camera.position.x += sx * 0.4; camera.position.y += sy * 0.4;
    camera.quaternion.copy(playerCar.quaternion);
    if (mode === 'T-CAM') camera.rotateX(-0.07);
    baseFov = (mode === 'T-CAM' ? 76 : 82) + speedFrac * 6;
  }
  if (Math.abs(camera.fov - baseFov) > 0.05) {
    camera.fov += (baseFov - camera.fov) * Math.min(1, 5 * dt);
    camera.updateProjectionMatrix();
  }
}

// --- resize -----------------------------------------------------------------------------------
window.addEventListener('resize', () => {
  const w = window.innerWidth, h = window.innerHeight;
  camera.aspect = w / h;
  camera.updateProjectionMatrix();
  renderer.setSize(w, h);
  composer.setSize(w, h);
});

// --- main loop: fixed-step physics, per-frame presentation ---------------------------------------
const clock = new THREE.Clock();
let accumulator = 0;
let carState = {
  speedKmh: 0, gear: 1, rpm: 4200, drsAvailable: false, drsActive: false,
  isBraking: false, isSliding: false, hitIntensity: 0, onCurb: false, u: 0,
};
// aiRacer hook: let raceState = null;

renderer.setAnimationLoop(() => {
  const dt = Math.min(clock.getDelta(), 0.1);
  const elapsed = clock.elapsedTime;

  buildControls(dt);

  accumulator += dt;
  let steps = 0;
  while (accumulator >= STEP && steps < MAX_SUBSTEPS) {
    carState = updateCarPhysics(playerCar, controls, trackData, STEP);
    // aiRacer hook: raceState = updateAIRacers(aiList, trackData, STEP);
    accumulator -= STEP;
    steps++;
  }
  if (steps === MAX_SUBSTEPS) accumulator = 0; // spiral-of-death guard

  // sun shadow frustum follows the player (crisp local shadows)
  sun.position.copy(playerCar.position).addScaledVector(sunDir, 380);
  sun.target.position.copy(playerCar.position);
  sun.target.updateMatrixWorld();

  // tunnel atmosphere: fog / exposure / bloom crossfade while inside
  const inTunnel = carState.u >= tunnel.uStart && carState.u <= tunnel.uEnd;
  const k = Math.min(1, 3 * dt);
  scene.fog.color.lerp(inTunnel ? TUNNEL_FOG : DAY_FOG, k);
  scene.fog.density += ((inTunnel ? 0.006 : 0.00135) - scene.fog.density) * k;
  renderer.toneMappingExposure += ((inTunnel ? 0.85 : 1.05) - renderer.toneMappingExposure) * k;
  bloom.strength += ((inTunnel ? 0.6 : 0.32) - bloom.strength) * k;

  // living harbor
  waterMat.uniforms.uTime.value = elapsed;

  updateCamera(dt, carState, elapsed);

  // audioHud hooks (Step 3):
  // updateAudio(carState);
  // updateHUD(raceState);

  composer.render();
});
